var (
        ErrNotFound = errors.New("employee not found")
        ErrInvalid  = errors.New("invalid employee")
)

type Employee struct {
        ID       string  `json:"id"`
        FullName string  `json:"full_name"`
        Position string  `json:"position"`
        Salary   float64 `json:"salary"`
        HiredOn  int64   `json:"hired_on"`
}

type EmployeeRepository interface {
        Create(Employee) (string, error)
        GetByID(string) (Employee, error)
        ChangePosition(string, string) error
}

type MemoryRepository struct {
        mu        sync.RWMutex
        employees map[string]Employee
        nextID    uint64
}

func NewMemoryRepository() *MemoryRepository {
        return &MemoryRepository{employees: make(map[string]Employee)}
}

func (r *MemoryRepository) Create(e Employee) (string, error) {
        if strings.TrimSpace(e.FullName) == "" || strings.TrimSpace(e.Position) == "" || e.Salary < 0 {
                return "", ErrInvalid
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        r.nextID++
        e.ID = formatID(r.nextID)
        if e.HiredOn == 0 {
                e.HiredOn = time.Now().Unix()
        }
        r.employees[e.ID] = e
        return e.ID, nil
}

func (r *MemoryRepository) GetByID(id string) (Employee, error) {
        r.mu.RLock()
        defer r.mu.RUnlock()
        e, ok := r.employees[id]
        if !ok {
                return Employee{}, ErrNotFound
        }
        return e, nil
}

func (r *MemoryRepository) ChangePosition(id, position string) error {
        if strings.TrimSpace(position) == "" {
                return ErrInvalid
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        e, ok := r.employees[id]
        if !ok {
                return ErrNotFound
        }
        e.Position = position
        r.employees[id] = e
        return nil
}

func formatID(n uint64) string {
        const digits = "0123456789"
        if n == 0 { return "0" }
        var buf [20]byte
        i := len(buf)
        for n > 0 {
                i--
                buf[i] = digits[n%10]
                n /= 10
        }
        return string(buf[i:])
}

import (

        "github.com/go-kit/kit/endpoint"
)

type API struct {
        createEndpoint        endpoint.Endpoint
        getByIDEndpoint       endpoint.Endpoint
        changePositionEndpoint endpoint.Endpoint
        logger                *log.Logger
}

func NewHTTPHandler(service EmployeeService, logger *log.Logger) http.Handler {
        api := &API{
                createEndpoint: makeCreateEndpoint(service),
                getByIDEndpoint: makeGetByIDEndpoint(service),
                changePositionEndpoint: makeChangePositionEndpoint(service),
                logger: logger,
        }
        mux := http.NewServeMux()
        mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
                path := strings.Trim(r.URL.Path, "/")
                switch {
                case path == "healthz" && r.Method == http.MethodGet:
                        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
                case path == "employees" && r.Method == http.MethodPost:
                        api.create(w, r)
                case strings.HasPrefix(path, "employees/") && r.Method == http.MethodGet:
                        r = withEmployeeID(r, strings.TrimPrefix(path, "employees/"))
                        api.getByID(w, r)
                case strings.HasPrefix(path, "employees/") && strings.HasSuffix(path, "/position") && r.Method == http.MethodPatch:
                        id := strings.TrimSuffix(strings.TrimPrefix(path, "employees/"), "/position")
                        r = withEmployeeID(r, id)
                        api.changePosition(w, r)
                default:
                        writeError(w, http.StatusNotFound, "route not found")
                }
        })
        return requestLogging(logger, mux)
}

func (a *API) create(w http.ResponseWriter, r *http.Request) {
        var e Employee
        if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, 1<<20)).Decode(&e); err != nil {
                writeError(w, http.StatusBadRequest, "invalid JSON body")
                return
        }
        response, err := a.createEndpoint(r.Context(), e)
        if err != nil {
                writeDomainError(w, err)
                return
        }
        writeJSON(w, http.StatusCreated, response)
}

func (a *API) getByID(w http.ResponseWriter, r *http.Request) {
        response, err := a.getByIDEndpoint(r.Context(), map[string]interface{}{"id": employeeID(r)})
        if err != nil {
                writeDomainError(w, err)
                return
        }
        writeJSON(w, http.StatusOK, response)
}

func (a *API) changePosition(w http.ResponseWriter, r *http.Request) {
        var body struct { Position string `json:"position"` }
        if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, 1<<20)).Decode(&body); err != nil {
                writeError(w, http.StatusBadRequest, "invalid JSON body")
                return
        }
        response, err := a.changePositionEndpoint(r.Context(), map[string]interface{}{
                "id": employeeID(r), "position": body.Position,
        })
        if err != nil {
                writeDomainError(w, err)
                return
        }
        writeJSON(w, http.StatusOK, response)
}

func writeDomainError(w http.ResponseWriter, err error) {
        switch {
        case errors.Is(err, ErrNotFound):
                writeError(w, http.StatusNotFound, err.Error())
        case errors.Is(err, ErrInvalid):
                writeError(w, http.StatusBadRequest, err.Error())
        default:
                writeError(w, http.StatusInternalServerError, "internal server error")
        }
}

func writeError(w http.ResponseWriter, status int, message string) {
        writeJSON(w, status, map[string]string{"error": message})
}

func writeJSON(w http.ResponseWriter, status int, value any) {
        var payload bytes.Buffer
        if err := json.NewEncoder(&payload).Encode(value); err != nil {
                http.Error(w, "internal server error", http.StatusInternalServerError)
                return
        }
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        if _, err := w.Write(payload.Bytes()); err != nil {
                log.Printf("failed to write HTTP response: %v", err)
        }
}

type employeeIDKey struct{}

func withEmployeeID(r *http.Request, id string) *http.Request {
        return r.WithContext(context.WithValue(r.Context(), employeeIDKey{}, id))
}

func employeeID(r *http.Request) string {
        id, _ := r.Context().Value(employeeIDKey{}).(string)
        return id
}

func requestLogging(logger *log.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                next.ServeHTTP(w, r)
                if logger != nil { logger.Printf("method=%s path=%s duration=%s", r.Method, strings.TrimSpace(r.URL.Path), time.Since(start)) }
        })
}


package main

import (
        "context"
        "errors"
        "flag"
        "log"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "google.golang.org/grpc"
)

func main() {
        httpAddr := flag.String("http.addr", ":8080", "HTTP listen address")
        grpcAddr := flag.String("grpc.addr", ":9090", "gRPC listen address")
        flag.Parse()

        logger := log.New(os.Stdout, "employee-service ", log.LstdFlags|log.LUTC)
        repo := NewMemoryRepository()
        service := NewEmployeeService(repo)

        grpcListener, err := listenGRPC(*grpcAddr)
        if err != nil {
                logger.Fatalf("gRPC listen: %v", err)
        }
        grpcServer := grpc.NewServer()
        registerEmployeeGRPCServer(grpcServer, newEmployeeGRPCServer(service))

        httpServer := &http.Server{
                Addr: *httpAddr, Handler: NewHTTPHandler(service, logger),
                ReadHeaderTimeout: 5 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        go func() {
                logger.Printf("HTTP server listening on %s", *httpAddr)
                if err := httpServer.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
                        logger.Printf("HTTP server: %v", err)
                        stop()
                }
        }()
        go func() {
                logger.Printf("gRPC server listening on %s", *grpcAddr)
                if err := grpcServer.Serve(grpcListener); err != nil {
                        logger.Printf("gRPC server: %v", err)
                        stop()
                }
        }()

        <-ctx.Done()
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        if err := httpServer.Shutdown(shutdownCtx); err != nil {
                logger.Printf("HTTP graceful shutdown failed: %v", err)
        }
        grpcDone := make(chan struct{})
        go func() {
                grpcServer.GracefulStop()
                close(grpcDone)
        }()
        select {
        case <-grpcDone:
        case <-shutdownCtx.Done():
                grpcServer.Stop()
        }
}
